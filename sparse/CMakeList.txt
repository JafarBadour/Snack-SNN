cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(Parallel-Dynamic-Sparse-Training)

# Set CMAKE_PREFIX_PATH to your libtorch installation directory
# Example: set(CMAKE_PREFIX_PATH "$ENV{HOME}/Downloads/libtorch/")
# Or set via environment variable: export CMAKE_PREFIX_PATH=/path/to/libtorch
if(DEFINED ENV{CMAKE_PREFIX_PATH})
    set(CMAKE_PREFIX_PATH "$ENV{CMAKE_PREFIX_PATH}")
endif()

find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
set(CMAKE_BUILD_TYPE "RELEASE")


add_executable(Parallel-Dynamic-Sparse-Training mult/main.cpp)
target_link_libraries(Parallel-Dynamic-Sparse-Training "${TORCH_LIBRARIES}")
set_property(TARGET Parallel-Dynamic-Sparse-Training PROPERTY CXX_STANDARD 17)
